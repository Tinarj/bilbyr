Synthetic data for projection pursuit

Generating recoverable geometric structures hidden in high-dimensional data

1 Generating synthetic datasets for projection pursuit

The preceding noise-sensitivity experiment added Gaussian measurement noise directly to a two-dimensional pattern. This experiment does not add measurement noise to the structure. Instead, it embeds a clean nonlinear structure into a higher-dimensional dataset by rotating each signal coordinate with an independent Gaussian nuisance coordinate. We study whether the original pattern remains visible in the observed coordinate pairs as the rotation angle changes, how their Stringy05 values change, and whether the known inverse rotation recovers the initial pattern and its index value.

Four patterns are considered: cubic, quartic, sine, and two moons. Each dataset has n = 500 observations and p = 10 variables: four potentially signal-bearing mixed coordinates and six additional independent Gaussian nuisance variables. We investigate \(\theta=0,\pi/12,\pi/6,\pi/4\), corresponding to 0°, 15°, 30°, and 45°. The same random seed is used within each pattern, so changing the angle does not change the underlying structure or nuisance draws.

2 Data generator and experimental settings

2.1 Data-generation mechanism

For the cubic, quartic, and sine examples, draw \(t\sim N(0,1)\) and set the first signal coordinate to \(t\). The second signal coordinate is, respectively, the third or fourth orthogonal-polynomial basis vector produced by R’s poly() function, or \(\sin(\pi t)\). In particular, poly(t, degree = 3)[, 3] is not the same as the raw function \(t^3\). For two moons, points are generated on two semicircular arcs.

Each signal coordinate is standardized using its sample mean and sample standard deviation:

\[ \widetilde{s}_j = \frac{s_j-\overline{s}_j}{\operatorname{sd}(s_j)}, \qquad j=1,2. \]

Consequently, the two signal coordinates have sample SD exactly 1. Generate independent nuisance coordinates \(n_1,n_2\sim N(0,1)\), which have SD 1.

The signal and nuisance coordinates are mixed by an orthogonal rotation:

\[ \begin{aligned} x_1 &= \cos(\theta)s_1+\sin(\theta)n_1, & x_3 &= -\sin(\theta)s_1+\cos(\theta)n_1,\\ x_2 &= \cos(\theta)s_2+\sin(\theta)n_2, & x_4 &= -\sin(\theta)s_2+\cos(\theta)n_2. \end{aligned} \]

Six additional nuisance variables \(x_5,\ldots,x_{10}\sim N(0,1)\) are generated independently. With independent signal and nuisance coordinates, the rotated variables have approximately unit sample variance.

2.2 Reproducible setup

Code 
library(tidyverse)
library(cassowaryr)

gen_hidden_stringy_components <- function(
    n = 500, p = 10, pattern = "cubic",
    theta = pi / 4, seed = 918
) {
  pattern <- match.arg(pattern, c("cubic", "quartic", "sine", "two_moons"))
  stopifnot(n >= 5, p >= 4)
  set.seed(seed)

  if (pattern == "two_moons") {
    group <- sample(rep(1:2, length.out = n))
    angle <- runif(n, 0, pi)
    s1 <- ifelse(group == 1, cos(angle), 1 - cos(angle))
    s2 <- ifelse(group == 1, sin(angle), 0.5 - sin(angle))
  } else {
    t <- rnorm(n)
    s1 <- t
    s2 <- switch(
      pattern,
      cubic = poly(t, degree = 3)[, 3],
      quartic = poly(t, degree = 4)[, 4],
      sine = sin(pi * t)
    )
  }


  s1 <- as.numeric(scale(s1))
  s2 <- as.numeric(scale(s2))

  n1 <- rnorm(n, sd = 1)
  n2 <- rnorm(n, sd = 1)

  x1 <-  cos(theta) * s1 + sin(theta) * n1
  x3 <- -sin(theta) * s1 + cos(theta) * n1
  x2 <-  cos(theta) * s2 + sin(theta) * n2
  x4 <- -sin(theta) * s2 + cos(theta) * n2

  observed <- tibble(x1 = x1, x2 = x2, x3 = x3, x4 = x4)
  if (p > 4) {
    extra_noise <- matrix(rnorm(n * (p - 4), sd = 1), nrow = n)
    colnames(extra_noise) <- paste0("x", 5:p)
    observed <- bind_cols(observed, as_tibble(extra_noise))
  }

  list(
    observed = observed,
    original = tibble(s1 = s1, s2 = s2)
  )
}

gen_hidden_stringy <- function(n = 500, p = 10, pattern = "cubic",
                              theta = pi / 4, seed = 918) {
  gen_hidden_stringy_components(n, p, pattern, theta, seed)$observed
}

n <- 500
p <- 10
seed <- 918
patterns <- c("cubic", "quartic", "sine", "two_moons")
angles <- tibble(
  angle = c(0, pi / 12, pi / 6, pi / 4),
  angle_label = factor(
    c("0°", "15°", "30°", "45°"),
    levels = c("0°", "15°", "30°", "45°")
  )
)

experiments <- tidyr::crossing(
  pattern = factor(patterns, levels = patterns),
  angles
) |>
  mutate(result = purrr::map2(
    as.character(pattern), angle,
    ~ gen_hidden_stringy_components(
      n = n, p = p, pattern = .x, theta = .y, seed = seed
    )
  ))

4 Original and mixed views

The observed-coordinate views among the four mixed variables are all six combinations:

\[ (x_1,x_2),\ (x_1,x_3),\ (x_1,x_4),\ (x_2,x_3),\ (x_2,x_4),\ (x_3,x_4). \]

Code 
observed_pairs <- combn(paste0("x", 1:4), 2, simplify = FALSE)

make_views <- function(result, theta) {
  x <- result$observed
  original <- result$original

  mixed <- purrr::map_dfr(observed_pairs, function(vars) {
    tibble(
      u = x[[vars[1]]],
      v = x[[vars[2]]],
      view = paste(vars, collapse = "–"),
      type = "Mixed pair"
    )
  })

  bind_rows(
    tibble(
      u = original$s1,
      v = original$s2,
      view = "Original s1–s2",
      type = "Original"
    ),
    mixed
  )
}

view_data <- experiments |>
  mutate(views = map2(result, angle, make_views)) |>
  select(pattern, angle, angle_label, views) |>
  unnest(views) |>
  mutate(
    view = factor(
      view,
      levels = c(
        "Original s1–s2",
        vapply(
          observed_pairs,
          paste,
          character(1),
          collapse = "–"
        )
      )
    )
  )

5 Calculate Stringy05 index values

We calculate Stringy05 for the original structure and each of the six mixed coordinate pairs.

Code 
#| label: score-views
rescale_scores <- TRUE

score_stringy05 <- function(x, y, rescale = rescale_scores) {
  as.numeric(
    cassowaryr::sc_stringy05(
      x = x,
      y = y,
      rescale = rescale
    )
  )
}

index_scores <- view_data |>
  group_by(pattern, angle, angle_label, view, type) |>
  summarise(
    value = score_stringy05(u, v),
    .groups = "drop"
  )

stopifnot(all(is.finite(index_scores$value)))

Let’s investigate smaller rotation angles between 0° and 15° to determine how quickly Stringy05 decreases when the original structure is mixed with Gaussian nuisance variables.

Code 
small_angles <- c(0, 1, 2, 3, 5, 7, 10, 12, 14)

experiments_small <- tidyr::crossing(
  pattern = patterns,
  angle = small_angles
) |>
  mutate(
    theta = angle * pi / 180,
    angle_label = factor(
      paste0(angle, "°"),
      levels = paste0(small_angles, "°")
    ),
    result = purrr::map2(
      pattern,
      theta,
      \(p, th) {

        # Use the same generator as in the original experiment
        x <- gen_hidden_stringy(
          n = 500,
          p = 10,
          pattern = p,
          theta = th,
          seed = 918
        )

        # Recover the original standardized coordinates
        # to construct the reference original view
        original <- tibble(
          s1 = cos(th) * x$x1 - sin(th) * x$x3,
          s2 = cos(th) * x$x2 - sin(th) * x$x4
        )

        list(
          observed = x,
          original = original
        )
      }
    )
  )

view_data_small <- experiments_small |>
  mutate(
    views = map2(result, angle, make_views)
  ) |>
  select(pattern, angle, angle_label, views) |>
  unnest(views) |>
  mutate(
    view = factor(
      view,
      levels = c(
        "Original s1–s2",
        vapply(
          observed_pairs,
          paste,
          character(1),
          collapse = "–"
        )
      )
    )
  )
